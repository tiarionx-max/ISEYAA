Fix two confirmed deployment blockers and add Swagger docs across the 5 gRPC-extracted microservice apps under `backend/apps/{news,waitlist,reviews,notifications,delivery-otp}-service`.

1. All 5 services currently bind their internal gRPC transport to `0.0.0.0:<port>` (IPv4-only). Railway's private network (`<name>.railway.internal`) is IPv6-only, so inter-service gRPC calls to any of these 5 services cannot resolve/connect once deployed — this is a deploy-blocking bug, not yet hit in production because these services haven't been cut over to live traffic.

Also add Swagger (`@nestjs/swagger`, already a backend dependency) to all 5 services' `main.ts`, documenting that their HTTP surface is health-only by design — their real API surface is gRPC, not REST.

```typescript
import { NestFactory } from '@nestjs/core';
import { MicroserviceOptions, Transport } from '@nestjs/microservices';
import { HealthImplementation, protoPath as healthCheckProtoPath } from 'grpc-health-check';
import { join } from 'path';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  app.connectMicroservice<MicroserviceOptions>({
    transport: Transport.GRPC,
    options: {
      package: 'news',
      protoPath: [
        join(__dirname, '../../../../../packages/proto/news.proto'),
        healthCheckProtoPath,
      ],
      url: '0.0.0.0:5009',
      onLoadPackageDefinition: (() => {
        let registered = false;
        return (pkg, server) => {
          if (registered) return;
          registered = true;
          const healthImpl = new HealthImplementation({ '': 'UNKNOWN' });
          healthImpl.addToServer(server);
          healthImpl.setStatus('', 'SERVING');
        };
      })(),
    },
  });

  await app.startAllMicroservices();
  await app.listen(process.env.PORT ?? 8080);
  console.log('news-service gRPC :5009, HTTP healthz :8080');
}

bootstrap();
```
