<!-- D-02's literal status filter, applied via LEFT JOIN back to the two
     status-bearing source tables (VisitorLog itself has no status column —
     D-07). Events rows need no status join: the write happens at the real
     physical scan moment (checkin()), so every EVENT row is already "valid"
     the instant it exists. D-02's prose says "not cancelled/refunded" —
     honor both BookingStatus values, not just CANCELLED literally. -->
```sql
SELECT v."lgaId", l.name AS "lgaName", TO_CHAR(v."visitedAt", 'YYYY-MM') AS month,
       v."userRole", COUNT(*)::int AS count
FROM visitor_logs v
LEFT JOIN lgas l ON v."lgaId" = l.id
LEFT JOIN bookings b ON v."sourceType" = 'STAY' AND v."sourceId" = b.id
LEFT JOIN tour_bookings tb ON v."sourceType" = 'TOUR' AND v."sourceId" = tb.id
WHERE v."visitedAt" <= NOW()
  AND (v."sourceType" != 'STAY' OR b.status NOT IN ('CANCELLED', 'REFUNDED'))
  AND (v."sourceType" != 'TOUR' OR tb.status NOT IN ('CANCELLED', 'REFUNDED'))
  -- + optional from/to/lgaId Prisma.sql conditional fragments (Prisma.empty when absent)
GROUP BY v."lgaId", l.name, month, v."userRole"
ORDER BY month ASC, count DESC
```
