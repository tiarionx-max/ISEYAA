```typescript
// VisitorEntriesChart — bucket userRole into 3 groups per D-04 ("TOURIST vs
// CITIZEN vs other"), sum counts per (lgaName, roleBucket), summing across
// every month present in the current from/to-filtered result set (the date
// range is already controlled by the page-level filter above the chart, so
// no separate month axis is needed here — LGA + role are this chart's two
// rendered dimensions).
type VisitorEntryRow = { lgaName: string; month: string; userRole: string; count: number };
type RoleBucket = 'TOURIST' | 'CITIZEN' | 'OTHER';
function bucketRole(role: string): RoleBucket {
  return role === 'TOURIST' || role === 'CITIZEN' ? role : 'OTHER';
}
function aggregateByLgaAndRole(data: VisitorEntryRow[]) {
  const byLga = new Map<string, { lgaName: string; TOURIST: number; CITIZEN: number; OTHER: number }>();
  for (const row of data) {
    const entry = byLga.get(row.lgaName) ?? { lgaName: row.lgaName, TOURIST: 0, CITIZEN: 0, OTHER: 0 };
    entry[bucketRole(row.userRole)] += row.count;
    byLga.set(row.lgaName, entry);
  }
  return Array.from(byLga.values());
}
// Render 3 stacked <Bar dataKey="TOURIST" stackId="a" .../> series per LGA bar,
// all in the forest family (no gold — non-financial data), differentiated by
// opacity so the 3 role segments remain visually distinct without introducing
// a second color family: TOURIST rgba(26,107,60,0.85), CITIZEN
// rgba(26,107,60,0.55), OTHER rgba(26,107,60,0.3). Add a small <Legend/>
// (recharts default, styled to match the dark tooltip) so the 3 segments are
// identifiable — this is the UI-SPEC-proportionate way to surface D-04's role
// dimension without a new filter control or a second chart.

// PurposeBreakdownChart — no role dimension in this query's output (D-04 only
// applies to visitor-entry counts, not purpose-of-visit); its second
// dimension is `month`, not `userRole`. Sum-collapse across months so each
// purpose renders as exactly one bar (dedup fix, checker warning 2) — this
// chart intentionally stays single-dimension-on-screen (purpose only) since
// no locked decision requires a by-month view for purpose-of-visit.
type PurposeRow = { purpose: string; month: string; count: number };
function aggregateByPurpose(data: PurposeRow[]) {
  const byPurpose = new Map<string, number>();
  for (const row of data) {
    byPurpose.set(row.purpose, (byPurpose.get(row.purpose) ?? 0) + row.count);
  }
  return Array.from(byPurpose, ([purpose, count]) => ({ purpose, count }));
}
```
