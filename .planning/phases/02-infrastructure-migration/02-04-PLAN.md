```typescript
// CRITICAL: loaded via --require BEFORE main.ts
// Do NOT import NestJS modules here
import { NodeSDK } from '@opentelemetry/sdk-node';
import { getNodeAutoInstrumentations } from '@opentelemetry/auto-instrumentations-node';
import { OTLPTraceExporter } from '@opentelemetry/exporter-trace-otlp-http';

const sdk = new NodeSDK({
  traceExporter: new OTLPTraceExporter({
    url: process.env.OTEL_EXPORTER_OTLP_ENDPOINT,
    headers: { Authorization: `Basic ${process.env.GRAFANA_CLOUD_OTLP_TOKEN}` },
  }),
  instrumentations: [getNodeAutoInstrumentations()],
});

sdk.start();
process.on('SIGTERM', () => { sdk.shutdown().finally(() => process.exit(0)); });
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| App → Grafana Cloud | OTLP trace data sent over HTTPS with Basic auth token |
| App → Sentry | Error reports (with stack traces) sent to Sentry; may contain PII in span attributes |
| Railway → /api/v1/health | Railway pings health endpoint from its internal network; endpoint is unauthenticated by design |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-15 | Information Disclosure | OTel traces containing PII (phone, BVN) | mitigate | Do not set custom span attributes for phone/BVN/NIN; auto-instrumentation captures only HTTP method/URL/status, not request bodies |
| T-02-16 | Information Disclosure | Sentry error reports containing PII in stack traces | mitigate | Set Sentry beforeSend hook in Phase 6 to scrub phone numbers from event data; for now tracesSampleRate: 0.1 limits exposure |
| T-02-17 | Information Disclosure | Swagger UI exposed in production (CONCERNS.md T-06) | mitigate | Gated behind APP_ENV !== 'production' in this plan — fully resolved |
| T-02-18 | Spoofing | OTel endpoint accepting traces without auth | accept | Grafana Cloud OTLP endpoint requires valid Basic auth token; token stored in Infisical (Plan 06) |
| T-02-19 | Information Disclosure | GRAFANA_CLOUD_OTLP_TOKEN in env | mitigate | Stored in Infisical only; Railway service only receives INFISICAL_TOKEN to pull secrets |
| T-02-46 | Information Disclosure | /api/v1/health exposes internal service status | accept | Health check returns only { status: 'ok' } with no internal details; no database or service checks in this implementation |
</threat_model>
