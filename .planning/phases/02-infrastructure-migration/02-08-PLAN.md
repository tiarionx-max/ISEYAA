```typescript
import { NestFactory } from '@nestjs/core';
import { MicroserviceOptions, Transport } from '@nestjs/microservices';
import { join } from 'path';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.createMicroservice<MicroserviceOptions>(AppModule, {
    transport: Transport.GRPC,
    options: {
      package: 'auth',
      protoPath: join(__dirname, '../../../../packages/proto/auth.proto'),
      url: '0.0.0.0:5001',
    },
  });
  await app.listen();
  console.log('auth-service gRPC listening on :5001');
}
bootstrap();
```

GrpcMethod controller (PATTERNS.md lines 495-505):
```typescript
@Controller()
export class AuthGrpcController {
  constructor(private readonly authService: AuthService) {}

  @GrpcMethod('AuthService', 'ValidateToken')
  async validateToken(data: ValidateTokenRequest): Promise<ValidateTokenResponse> {
    const user = await this.authService.validateJwt(data.token);
    return { valid: !!user, userId: user?.id ?? '', role: user?.role ?? '' };
  }

  @GrpcMethod('AuthService', 'GetUser')
  async getUser(data: GetUserRequest): Promise<GetUserResponse> {
    const user = await this.authService.findById(data.userId);
    return { id: user.id, email: user.email, phone: user.phone, role: user.role, status: user.status };
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| API gateway → auth-service | gRPC call on Railway private network; carries JWT token for validation |
| API gateway → wallet-service | gRPC call carrying wallet_id and amount; SELECT FOR UPDATE required on debit |
| gRPC metadata | userId and role passed in metadata for authorization in internal services |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-30 | Spoofing | Rogue service calling wallet-service gRPC Credit endpoint | mitigate | Railway private network limits callers to same-project services; credit method validates caller context via metadata userId |
| T-02-31 | Tampering | gRPC Credit called with manipulated amount | mitigate | WalletService.creditWallet() validates amount > 0; SELECT FOR UPDATE prevents concurrent debit race; same protections as HTTP path |
| T-02-32 | Information Disclosure | JWT token transmitted in gRPC metadata | accept | gRPC calls are on Railway internal network (private DNS); not exposed to internet; TLS termination at Railway level |
| T-02-33 | Elevation of Privilege | auth-service gRPC port exposed publicly | mitigate | gRPC ports (5001, 5002) are internal only — not exposed in railway.toml as public ports; REST (3001) is the only public port |
</threat_model>
