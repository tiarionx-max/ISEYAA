    async getStatement(walletId: string, opts: { dateFrom?: string; dateTo?: string } = {}) {
      return this.prisma.transaction.findMany({
        where: {
          walletId,
          type: 'CREDIT',
          ...(opts.dateFrom || opts.dateTo
            ? { createdAt: { ...(opts.dateFrom && { gte: new Date(opts.dateFrom) }), ...(opts.dateTo && { lte: new Date(opts.dateTo) }) } }

        },
        orderBy: { createdAt: 'desc' },

    ```
    import { Controller, Get, Query, UseGuards, NotFoundException } from '@nestjs/common';
    import { ApiTags, ApiBearerAuth, ApiOperation, ApiQuery } from '@nestjs/swagger';
    import { JwtAuthGuard } from '../../modules/auth/guards/jwt-auth.guard';
    import { CurrentUser } from '../decorators/current-user.decorator';
    import { UserRole } from '../enums/user-role.enum';
    import { PrismaService } from '../../prisma/prisma.service';
    import { SettlementService } from '../services/settlement.service';

    @ApiTags('settlements')
    @ApiBearerAuth()
    @UseGuards(JwtAuthGuard)
    @Controller('settlements')
    export class SettlementController {
      constructor(private prisma: PrismaService, private settlementService: SettlementService) {}

      @Get('statement')
      @ApiOperation({ summary: 'Itemized settlement statement — self-scoped, admin-overridable by walletId' })
      @ApiQuery({ name: 'walletId', required: false, description: 'SUPER_ADMIN/LGA_ADMIN only — any other role\'s value is ignored' })
      @ApiQuery({ name: 'dateFrom', required: false })
      @ApiQuery({ name: 'dateTo', required: false })
      async getStatement(
        @CurrentUser() user: { userId: string; role: string },
        @Query('walletId') walletId?: string,
        @Query('dateFrom') dateFrom?: string,
        @Query('dateTo') dateTo?: string,
      ) {
        const isAdmin = user.role === UserRole.SUPER_ADMIN || user.role === UserRole.LGA_ADMIN;
        let targetWalletId: string;
        if (isAdmin && walletId) {
          targetWalletId = walletId;
        } else {
          const wallet = await this.prisma.wallet.findUnique({ where: { userId: user.userId }, select: { id: true } });
          if (!wallet) throw new NotFoundException('Wallet not found for current user');
          targetWalletId = wallet.id;
        }
        return this.settlementService.getStatement(targetWalletId, { dateFrom, dateTo });
      }
    }
    ```
    **Critical IDOR closure:** note the `isAdmin && walletId` condition — for ANY non-admin role,
    the `walletId` query param is completely bypassed and `targetWalletId` is always derived from
    `user.userId` via the wallet lookup. Do not add any code path that trusts a client-supplied
    `walletId` for a non-admin role, and do not add a separate "explicit consent" bypass.

    Register in `backend/src/common/common.module.ts`: import `SettlementController` from
    `./controllers/settlement.controller`, add it to the `controllers` array alongside
    `UploadController` (leave `providers`/`exports` untouched here — `SettlementService` is already
    registered by Plan 12-01).
