```typescript
import { Injectable } from '@nestjs/common';
import { v4 as uuidv4 } from 'uuid';

export type ReferencePrefix = 'FUND' | 'TKT' | 'STY' | 'ORD' | 'ESC' | 'SBO' | 'TOUR';

@Injectable()
export class ReferenceService {
  /** Standard 12-char reference: `ISY-${prefix}-${12char uppercase hex}`. */
  generate(prefix: ReferencePrefix): string {
    const tail = uuidv4().replace(/-/g, '').slice(0, 12).toUpperCase();
    return `ISY-${prefix}-${tail}`;
  }

  /** Escrow uses 8-char per CLAUDE.md naming convention. */
  generateEscrow(): string {
    const tail = uuidv4().replace(/-/g, '').slice(0, 8).toUpperCase();
    return `ISY-ESC-${tail}`;
  }
}
```

```typescript
async refundCharge(reference: string, amountKobo?: number, reason?: string): Promise<{
  id: string;
  amount: number;       // kobo
  status: string;       // 'pending' | 'processed' | 'failed'
}> {
  const secretKey = this.config.get<string>('PAYSTACK_SECRET_KEY', '');
  if (!secretKey) {
    this.logger.warn('PAYSTACK_SECRET_KEY not set — refund stubbed in dev');
    return { id: `stub_${reference}`, amount: amountKobo ?? 0, status: 'pending' };
  }
  const body: Record<string, any> = { transaction: reference };
  if (amountKobo) body.amount = amountKobo;
  if (reason) body.customer_note = reason;
  try {
    const { data } = await axios.post('https://api.paystack.co/transaction/refund', body, {
      headers: { Authorization: `Bearer ${secretKey}`, 'Content-Type': 'application/json' },
      timeout: 10_000,
    });
    return { id: String(data.data.id), amount: Number(data.data.amount), status: String(data.data.status) };
  } catch (err: any) {
    this.logger.error(`Paystack refund failed for ${reference}`, err?.response?.data ?? err.message);
    throw new ServiceUnavailableException('Refund gateway unavailable. Retry queued.');
  }
}
```

**Step 2:** Create `backend/src/common/services/refund.service.ts`:

```typescript
@Injectable()
export class RefundService {
  private readonly logger = new Logger(RefundService.name);

  constructor(
    private prisma: PrismaService,
    private paystack: PaystackService,
    private referenceService: ReferenceService,
  ) {}

  async refund(input: RefundInput): Promise<RefundResult> {
    const refundRef = `${input.paystackReference}-RFND`;

    // Idempotency: replay-safe — if we already wrote this REFUND row, return it.
    const existing = await this.prisma.transaction.findUnique({ where: { reference: refundRef } });
    if (existing) {
      return {
        refundReference: existing.reference,
        paystackRefundId: existing.gatewayRef ?? '',
        amountRefunded: Number(existing.amount),
        status: 'SUCCESS',
        transactionId: existing.id,
      };
    }

    const amountKobo = input.amountKobo;
    const amountNgn = amountKobo / 100;

    // 1. Call Paystack refund endpoint
    const paystackResult = await this.paystack.refundCharge(
      input.paystackReference,
      amountKobo,
      input.reason,
    );

    // 2. Write REFUND Transaction row on buyer wallet — SELECT FOR UPDATE for consistency
    const txn = await this.prisma.$transaction(async (tx) => {
      await tx.$executeRaw`SELECT id FROM wallets WHERE id = ${input.walletId} FOR UPDATE`;
      const wallet = await tx.wallet.findUnique({ where: { id: input.walletId } });
      if (!wallet) throw new NotFoundException('Wallet not found for refund');

      // Note: this REFUND row records the gateway-side reversal. It does NOT touch wallet.balance
      // (the original charge never credited the buyer wallet — Paystack returns money to the card).
      // We write a zero-balance-delta REFUND row purely as a ledger marker for audit + idempotency.
      const balance = Number(wallet.balance);
      return tx.transaction.create({
        data: {
          walletId: input.walletId,
          type: 'REFUND',
          status: paystackResult.status === 'processed' ? 'SUCCESS' : 'PENDING',
          amount: amountNgn,
          currency: 'NGN',
          reference: refundRef,
          gateway: 'PAYSTACK',
          gatewayRef: paystackResult.id,
          description: input.reason ?? 'Refund issued',
          balanceBefore: balance,
          balanceAfter: balance,                     // unchanged — money goes back to card not wallet
          metadata: { ...input.metadata, module: 'refund', originalReference: input.paystackReference },
        },
      });
    });

    return {
      refundReference: refundRef,
      paystackRefundId: paystackResult.id,
      amountRefunded: amountNgn,
      status: paystackResult.status === 'processed' ? 'SUCCESS' : 'PENDING',
      transactionId: txn.id,
    };
  }
}
```

Important behavioral notes for the executor:
- The REFUND Transaction has `balanceAfter === balanceBefore` because Paystack returns the money to the original card, not to the user's in-app wallet. The row exists purely as a ledger marker (for audit + idempotency check).
- If a future flow needs to credit the wallet instead of the card (e.g. promo refund), that's a separate `creditWallet` call — NOT this service's job.
- Idempotency is keyed on `${paystackReference}-RFND` — a single original Paystack reference can have at most one refund record from this service.
