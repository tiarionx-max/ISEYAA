Stand up TourPackage CRUD + listing + admin approval. Closes TOUR-02 + TOUR-03 (the search/filter half) + TOUR-08 (admin queue) for the backend. Date-constraint check for tourist bookings is implemented in plan 09-05; this plan handles the package side of the contract.

**Critical guards baked into createTourPackage:**

1. **Guide must be APPROVED** — inject `TourGuideService` (exported by 09-03) and call `findByIdInternal(tourGuideId)`. If null or `status !== 'APPROVED'`, throw `ForbiddenException('Only approved tour guides can list packages')`. The caller's `userId` must also match the guide's `userId` (a guide can only create packages for themselves).
2. **At least 1 attraction** — `attractionIds.length >= 1`, and every ID must resolve to a non-deleted Attraction row. Throw `BadRequestException` with the missing ID.
3. **Optional propertyId resolves** — if present, must resolve to a non-deleted Property row.
4. **Optional eventIds resolve** — each must resolve to a non-deleted Event row.
5. **settlementSplit sum <= 100** — sum the `percentage` values; if > 100, throw `BadRequestException('Settlement split exceeds 100%')`. The DB CHECK constraint from 09-01 is the second line of defence; this is the user-friendly first line.
6. **Each settlementSplit entry's vendorId resolves** — for vendorType=GUIDE the vendorId must be a TourGuide ID; for HOST a Property.hostId; for ORGANISER an Event.organizerId; for ATTRACTION an Attraction ID. Resolution rules:
   - GUIDE → match `tourGuide.id`, confirm `userId` (wallet-bearing entity).
   - HOST → `property.hostId` must point to a User with a Wallet.
   - ORGANISER → `event.organizerId` must point to a User with a Wallet.
   - ATTRACTION → resolve to `tour.government_wallet_user_id` PlatformConfig at booking time (NOT here). Service validates the attraction exists but does NOT require a Wallet row for the attraction itself. CONTEXT §Specifics flags this v1 behavior.
7. **Category is a TourPackageCategory enum value** — class-validator `@IsEnum`.
8. **itineraryTemplate.length >= 1** — and each entry has `{ hour: int 0-23, title: non-empty, description: string }`.

**Edit lifecycle (v1):**
- DRAFT → fully editable by owning guide.
- PENDING → editable, returns to DRAFT on edit (no auto-revert; guide must explicitly re-submit).
- APPROVED → BLOCKED for direct edit (return 409). CONTEXT §deferred §versioning calls out snapshot-on-booking as the v1 stand-in; full versioning is deferred. The guide must clone the package to make changes (clone endpoint = future work; for now, surface the message in the 409 response).
- REJECTED → editable (returns to DRAFT on edit).

**Slug generation** — same pattern as `marketplace.service.slugify` (read it): `${slugify(name)}-${uuidv4().slice(0,8)}`. The 8-char suffix prevents collisions across guides with same package name.

@Injectable()
export class TourPackageService {
  private readonly logger = new Logger(TourPackageService.name);

  constructor(
    private prisma: PrismaService,
    private tourGuides: TourGuideService,   // imported from 09-03 module
  ) {}

  async create(actorUserId: string, dto: CreateTourPackageDto): Promise<TourPackage> {
    // 1. Guide must be APPROVED + owned by the actor
    const guide = await this.tourGuides.findByIdInternal(dto.tourGuideId);
    if (!guide) throw new NotFoundException('Tour guide not found');
    if (guide.userId !== actorUserId) throw new ForbiddenException('You can only create packages for your own guide profile');
    if (guide.status !== 'APPROVED') throw new ForbiddenException('Only approved tour guides can list packages');

    // 2. Validate attractions resolve (1+ required is already DTO-enforced)
    const attractions = await this.prisma.attraction.findMany({
      where: { id: { in: dto.attractionIds }, deletedAt: null },
      select: { id: true },
    });
    if (attractions.length !== dto.attractionIds.length) {
      const missing = dto.attractionIds.filter(id => !attractions.find(a => a.id === id));
      throw new BadRequestException(`Unknown attraction(s): ${missing.join(', ')}`);
    }

    // 3. Optional property resolves
    if (dto.propertyId) {
      const property = await this.prisma.property.findFirst({
        where: { id: dto.propertyId, deletedAt: null },
        select: { id: true, hostId: true },
      });
      if (!property) throw new BadRequestException(`Unknown property: ${dto.propertyId}`);
    }

    // 4. Optional events resolve
    if (dto.eventIds?.length) {
      const events = await this.prisma.event.findMany({
        where: { id: { in: dto.eventIds }, deletedAt: null },
        select: { id: true },
      });
      if (events.length !== dto.eventIds.length) {
        throw new BadRequestException('One or more event IDs are unknown');
      }
    }

    // 5. settlementSplit sum guard (DB CHECK is the second line; this is user-friendly)
    const splitSum = dto.settlementSplit.reduce((s, e) => s + e.percentage, 0);
    if (splitSum > 100) throw new BadRequestException(`Settlement split sums to ${splitSum}% — must be ≤ 100`);

    // 6. Validate each settlementSplit entry's vendorId resolves to a settable wallet anchor
    await this.validateSplitEntries(dto.settlementSplit);

    // 7. itineraryTemplate hour ordering sanity (DTO covers shape; we add cross-field check)
    const hours = dto.itineraryTemplate.map(i => i.hour);
    if (!hours.every((h, i, arr) => i === 0 || h >= arr[i - 1])) {
      throw new BadRequestException('itineraryTemplate must be ordered by ascending hour');
    }

    const slug = `${slugify(dto.name)}-${uuidv4().slice(0, 8)}`;

    return this.prisma.tourPackage.create({
      data: {
        slug,
        lgaId: dto.lgaId,
        tourGuideId: dto.tourGuideId,
        name: dto.name,
        description: dto.description,
        category: dto.category,
        price: dto.price,
        durationHours: dto.durationHours,
        maxGroupSize: dto.maxGroupSize,
        coverImageUrl: dto.coverImageUrl,
        imageUrls: dto.imageUrls ?? [],
        attractionIds: dto.attractionIds,
        propertyId: dto.propertyId ?? null,
        eventIds: dto.eventIds ?? [],
        transportNote: dto.transportNote,
        itineraryTemplate: dto.itineraryTemplate as any,
        settlementSplit: dto.settlementSplit as any,
        status: 'DRAFT',
      },
    });
  }

  private async validateSplitEntries(entries: SettlementSplitEntryDto[]) {
    for (const e of entries) {
      switch (e.vendorType) {
        case 'GUIDE': {
          const g = await this.prisma.tourGuide.findFirst({ where: { id: e.vendorId, deletedAt: null } });
          if (!g) throw new BadRequestException(`Settlement split: unknown tour guide ${e.vendorId}`);
          break;
        }
        case 'HOST': {
          const p = await this.prisma.property.findFirst({
            where: { id: e.vendorId, deletedAt: null },
            select: { hostId: true },
          });
          if (!p) throw new BadRequestException(`Settlement split: unknown property ${e.vendorId}`);
          if (!p.hostId) throw new BadRequestException(`Property ${e.vendorId} has no hostId — cannot route HOST commission`);
          // Wallet check (host must have a wallet for settlement)
          const w = await this.prisma.wallet.findUnique({ where: { userId: p.hostId } });
          if (!w) throw new BadRequestException(`Host wallet not found for property ${e.vendorId}`);
          break;
        }
        case 'ORGANISER': {
          const ev = await this.prisma.event.findFirst({
            where: { id: e.vendorId, deletedAt: null },
            select: { organizerId: true },
          });
          if (!ev) throw new BadRequestException(`Settlement split: unknown event ${e.vendorId}`);
          const w = await this.prisma.wallet.findUnique({ where: { userId: ev.organizerId } });
          if (!w) throw new BadRequestException(`Organiser wallet not found for event ${e.vendorId}`);
          break;
        }
        case 'ATTRACTION': {
          // V1 routes attraction commissions to PlatformConfig 'tour.government_wallet_user_id'.
          // CONTEXT §Specifics + §deferred §Attraction-as-vendor wallet — flagged for operator setup
          // at deploy time. Validate the attraction exists; do NOT require a Wallet here (settlement
          // engine 09-06 reads the government wallet ID and falls back to platform commission if unset).
          const a = await this.prisma.attraction.findFirst({ where: { id: e.vendorId, deletedAt: null } });
          if (!a) throw new BadRequestException(`Settlement split: unknown attraction ${e.vendorId}`);
          break;
        }
      }
    }
  }

  async findOwn(userId: string) {
    const guide = await this.prisma.tourGuide.findUnique({ where: { userId } });
    if (!guide) return [];
    return this.prisma.tourPackage.findMany({
      where: { tourGuideId: guide.id, deletedAt: null },
      orderBy: { createdAt: 'desc' },
    });
  }

  findAll(filters: {
    category?: TourPackageCategory;
    lgaId?: string;
    tourGuideId?: string;
    q?: string;
    page?: number;
    limit?: number;
  }) {
    const { category, lgaId, tourGuideId, q, page = 1, limit = 24 } = filters;
    return this.prisma.tourPackage.findMany({
      where: {
        deletedAt: null,
        status: 'APPROVED',     // PUBLIC list filters to APPROVED only
        ...(category && { category }),
        ...(lgaId && { lgaId }),
        ...(tourGuideId && { tourGuideId }),
        ...(q && { name: { contains: q, mode: 'insensitive' } }),
      },
      include: PACKAGE_PUBLIC_INCLUDE,
      orderBy: [{ isFeatured: 'desc' }, { createdAt: 'desc' }],
      skip: (page - 1) * limit,
      take: Math.min(limit, 50),
    });
  }

  async findBySlug(slug: string) {
    const pkg = await this.prisma.tourPackage.findFirst({
      where: { slug, status: 'APPROVED', deletedAt: null },
      include: PACKAGE_PUBLIC_INCLUDE,
    });
    if (!pkg) throw new NotFoundException('Tour package not found');

    const [attractions, property, events] = await Promise.all([
      this.prisma.attraction.findMany({ where: { id: { in: pkg.attractionIds }, deletedAt: null } }),
      pkg.propertyId ? this.prisma.property.findUnique({ where: { id: pkg.propertyId } }) : null,
      pkg.eventIds?.length ? this.prisma.event.findMany({ where: { id: { in: pkg.eventIds }, deletedAt: null } }) : [],
    ]);

    return { ...pkg, attractions, property, events };
  }

  async update(id: string, actorUserId: string, dto: UpdateTourPackageDto) {
    const pkg = await this.prisma.tourPackage.findFirst({ where: { id, deletedAt: null } });
    if (!pkg) throw new NotFoundException('Tour package not found');
    const guide = await this.prisma.tourGuide.findUnique({ where: { id: pkg.tourGuideId } });
    if (!guide || guide.userId !== actorUserId) throw new ForbiddenException('Not your package');
    if (pkg.status === 'APPROVED') {
      throw new ConflictException('Cannot edit an APPROVED package — clone it to create a new version (versioning is a deferred feature)');
    }

    // Re-run partial validation for any updated fields
    if (dto.attractionIds || dto.propertyId !== undefined || dto.eventIds || dto.settlementSplit) {
      const merged: CreateTourPackageDto = { ...(pkg as any), ...dto } as any;
      await this.validateSplitEntries(merged.settlementSplit ?? []);
    }

    return this.prisma.tourPackage.update({
      where: { id },
      data: {
        ...dto,
        // Edits to PENDING|REJECTED revert to DRAFT so the guide must explicitly re-submit
        status: pkg.status === 'PENDING' || pkg.status === 'REJECTED' ? 'DRAFT' : pkg.status,
      },
    });
  }

  async submitForReview(id: string, actorUserId: string) {
    const pkg = await this.prisma.tourPackage.findFirst({ where: { id, deletedAt: null } });
    if (!pkg) throw new NotFoundException('Tour package not found');
    const guide = await this.prisma.tourGuide.findUnique({ where: { id: pkg.tourGuideId } });
    if (!guide || guide.userId !== actorUserId) throw new ForbiddenException('Not your package');
    if (pkg.status !== 'DRAFT' && pkg.status !== 'REJECTED') {
      throw new ConflictException(`Cannot submit package in status ${pkg.status}`);
    }
    return this.prisma.tourPackage.update({ where: { id }, data: { status: 'PENDING' } });
  }

  async softDelete(id: string, actorUserId: string) {
    const pkg = await this.prisma.tourPackage.findFirst({ where: { id, deletedAt: null } });
    if (!pkg) throw new NotFoundException('Tour package not found');
    const guide = await this.prisma.tourGuide.findUnique({ where: { id: pkg.tourGuideId } });
    if (!guide || guide.userId !== actorUserId) throw new ForbiddenException('Not your package');
    if (pkg.status === 'APPROVED') throw new ConflictException('Cannot delete an APPROVED package with bookings — contact admin');
    await this.prisma.tourPackage.update({ where: { id }, data: { deletedAt: new Date() } });
    return { deleted: true };
  }

  findPendingQueue(opts: { page?: number; limit?: number }) {
    const { page = 1, limit = 24 } = opts;
    return this.prisma.tourPackage.findMany({
      where: { status: 'PENDING', deletedAt: null },
      include: PACKAGE_PUBLIC_INCLUDE,
      orderBy: { createdAt: 'asc' },
      skip: (page - 1) * limit,
      take: Math.min(limit, 50),
    });
  }

  async adminDecide(id: string, dto: AdminDecideTourPackageDto, actorId: string) {
    const pkg = await this.prisma.tourPackage.findFirst({ where: { id, deletedAt: null } });
    if (!pkg) throw new NotFoundException('Tour package not found');
    if (pkg.status !== 'PENDING') throw new ConflictException(`Cannot decide package in status ${pkg.status}`);
    return this.prisma.tourPackage.update({
      where: { id },
      data: {
        status: dto.decision,
        metadata: {
          ...((pkg.metadata as any) ?? {}),
          adminDecision: { decision: dto.decision, reason: dto.reason ?? null, actorId, at: new Date().toISOString() },
        },
      },
    });
  }

  /** INTERNAL — used by 09-05 booking validation */
  async findByIdInternal(id: string) {
    return this.prisma.tourPackage.findFirst({
      where: { id, deletedAt: null },
      select: {
        id: true, slug: true, status: true, tourGuideId: true,
        price: true, durationHours: true, maxGroupSize: true,
        attractionIds: true, propertyId: true, eventIds: true,
        itineraryTemplate: true, settlementSplit: true, name: true,
      },
    });
  }
}
